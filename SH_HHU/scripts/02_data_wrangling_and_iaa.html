Function for reading in Excel sheets

This function reads in the Excel sheet and then returns a dataframe with either the vocalizations or the ape gestures (where two sheets with apes are available in one workbook, both are combined).

# function for adding participant prefix to column name
add_participant_prefix <- function(x) {
  x <- unlist(strsplit(x, split = " "))
  x <- paste0(x[2], "_", x[1])
  return(x)
}

# function for replacing non-numeric ratings

replace_non_numerics <- function(df) {
  
  # find relevant columns
  find_ratings <- grep("Rating", colnames(df))

for(i in 1:length(find_ratings)) {
  if(!is.numeric(unlist(df[,find_ratings[i]]))) {
    df[,find_ratings[i]] <- suppressWarnings(as.numeric(unlist(df[,find_ratings[i]])))
  }
}
  
  return(df)
}
